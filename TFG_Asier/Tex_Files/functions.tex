\makeatletter

\newcommand{\clearemptydoublepage}{%
    \clearpage
    \if@twoside
        \ifodd\value{page}
            % The next page is already odd: no blank page is required.
        \else
            % Insert an empty even page so that the next page is odd.
            \thispagestyle{empty}%
            \null
            \newpage
        \fi
    \fi
}

\makeatother


\newcommand{\blankpage}{%
    \null
    \thispagestyle{empty}%
    \newpage
}

% Anexo numerado internamente, pero sin mostrar la letra en el título
\newcommand{\hiddenappendixchapter}[2]{%
    \refstepcounter{chapter}%
    \chapter*{#1}%
    \phantomsection%
    \addcontentsline{toc}{chapter}
        {\protect\numberline{\thechapter}#1}%
    \markboth{#1}{#1}%
    \label{#2}%
}
